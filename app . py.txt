import streamlit as st
import google.generativeai as genai

# Eikhane tomar API KEY bosao
genai.configure(api_key="TOMAR_API_KEY_EIKHANE")

model = genai.GenerativeModel("gemini-2.5-flash")

st.set_page_config(page_title="Amar Bondhu Bot")
st.title("🤖 Amar Bondhu Bot")
st.write("Hi! Amar sathe kotha bolo!")

if "chat" not in st.session_state:
    st.session_state.chat = []

for msg in st.session_state.chat:
    st.chat_message(msg["role"]).write(msg["text"])

if prompt := st.chat_input("Eikhane likho..."):
    st.session_state.chat.append({"role": "user", "text": prompt})
    st.chat_message("user").write(prompt)
    
    response = model.generate_content(prompt)
    st.session_state.chat.append({"role": "assistant", "text": response.text})
    st.chat_message("assistant").write(response.text)
